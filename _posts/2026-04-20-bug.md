<script>
  var fading_out_craig=false;
  function fade_out_craig(){
    fading_out_craig=true;
    var crg=document.getElementById("craigory");
    var op=parseFloat(crg.style.opacity);
    if (op==0){
      crg.style.display="none";
      fading_out_craig=false;
    }
    else{
      crg.style.opacity=Math.max(0,op-.05);
      setTimeout(fade_out_craig, 10);
    }
  }
  function craig(duration=100){
    if (document.documentElement.getAttribute("goose-setting") == "silly"){
      var crg=document.getElementById("craigory");
      crg.style.display="inherit";
      crg.style.opacity=1;
      if (!fading_out_craig){
        setTimeout(fade_out_craig, duration);
      }
    }
  }
  document.addEventListener("mousemove", craig);
  document.addEventListener("scroll", craig);
  document.addEventListener("DOMContentLoaded", craig);
</script>
