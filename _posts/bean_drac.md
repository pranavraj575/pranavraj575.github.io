let swtch = (tag) => {
  stuff=Array.from(document.getElementsByClassName("bean")).concat(
      Array.from(document.getElementsByClassName("drac"))).concat(
      Array.from(document.getElementsByClassName("both")));
  for (element of stuff){
    if (element.classList.contains(tag) || tag== "all"){
      reveal_img(element);
    } 
    else {
      hide_img(element);
    }
  }
  for (t of ['bean', 'drac', 'both', 'all']){
    document.getElementById(t+'-box').checked=false;
  }
  document.getElementById(tag+'-box').checked=true;
  title=document.getElementsByTagName('title')[0];
  if (tag=='bean' || tag=='drac'){
    title.innerHTML=" "+tag+" | Pranav Rajbhandari "
  } else {
    title.innerHTML=" bean &amp; drac | Pranav Rajbhandari "
  }
};

let hide_img = (element) => {
  if (element.tagName=="SPAN" || element.tagName=="DIV"){
    if (element.style.display && element.style.display !="none") {
      element.setAttribute('display-style',element.style.display)
    }
    element.style.display="none";
    for (child of element.children){
      hide_img(child);
    }
  }
  else if (element.tagName=="IMG"){
    if (element.classList.contains('spotlight')){
      element.classList.remove("spotlight");
      element.classList.add("notlight-bd");
    }
  }
  else {
    for (child of element.children){
      hide_img(child);
    }
  }
};

let reveal_img = (element) => {
  if ((element.tagName=="SPAN" || element.tagName=="DIV") && element.style.display =="none"){
    if (element.hasAttribute('display-style')) {
      element.style.display=element.getAttribute('display-style')
    }
    else{
      element.style.removeProperty('display');
    }
    for (child of element.children){
      reveal_img(child);
    }
  }
  else if (element.tagName=="IMG"){
    if (element.classList.contains('notlight-bd')){
      element.classList.remove("notlight-bd");
      element.classList.add("spotlight");
    }
  }
  else {
    for (child of element.children){
      reveal_img(child);
    }
  }
};

title=document.getElementsByClassName("post-title")[0];
